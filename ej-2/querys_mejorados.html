Querys del Esquema de Base de Datos

Sistema de Gestión de Biblioteca (SQLite)

Este documento contiene las consultas SQL para la creación y gestión de la base de datos de la biblioteca. Incluye la estructura de tablas normalizada, índices para optimización de consultas y triggers para automatizar la lógica de negocio.

1. Creación de Tablas (Estructura Avanzada)

Se habilita el uso de llaves foráneas y se definen las tablas con atributos extendidos (ISBN, ubicaciones, estados y fechas límite).


PRAGMA foreign_keys = ON;

-- Tabla: autor
CREATE TABLE autor (
    id_autor INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL,
    apellidos TEXT NOT NULL,
    nacionalidad TEXT
);

-- Tabla: editorial
CREATE TABLE editorial (
    id_editorial INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL,
    pais TEXT
);

-- Tabla: categoria
CREATE TABLE categoria (
    id_categoria INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL UNIQUE
);

-- Tabla: libro
CREATE TABLE libro (
    id_libro INTEGER PRIMARY KEY AUTOINCREMENT,
    titulo TEXT NOT NULL,
    id_autor INTEGER,
    id_editorial INTEGER,
    id_categoria INTEGER,
    sinopsis TEXT,
    FOREIGN KEY (id_autor) REFERENCES autor(id_autor) ON DELETE SET NULL,
    FOREIGN KEY (id_editorial) REFERENCES editorial(id_editorial) ON DELETE SET NULL,
    FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria) ON DELETE SET NULL
);

-- Tabla: lanzamiento_publicacion
CREATE TABLE lanzamiento_publicacion (
    id_lanzamiento INTEGER PRIMARY KEY AUTOINCREMENT,
    id_libro INTEGER NOT NULL,
    isbn TEXT UNIQUE,
    idioma TEXT DEFAULT 'Español',
    fecha_lanzamiento DATE,
    numero_edicion INTEGER,
    FOREIGN KEY (id_libro) REFERENCES libro(id_libro) ON DELETE CASCADE
);

-- Tabla: publicacion
CREATE TABLE publicacion (
    id_publicacion INTEGER PRIMARY KEY AUTOINCREMENT,
    id_lanzamiento INTEGER NOT NULL,
    codigo_inventario TEXT UNIQUE NOT NULL,
    ubicacion_estante TEXT,
    estado TEXT DEFAULT 'Disponible',
    FOREIGN KEY (id_lanzamiento) REFERENCES lanzamiento_publicacion(id_lanzamiento) ON DELETE CASCADE
);

-- Tabla: usuario
CREATE TABLE usuario (
    id_usuario INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL,
    apellidos TEXT NOT NULL,
    correo TEXT UNIQUE NOT NULL,
    telefono TEXT,
    estado TEXT DEFAULT 'Activo'
);

-- Tabla: prestamo
CREATE TABLE prestamo (
    id_prestamo INTEGER PRIMARY KEY AUTOINCREMENT,
    id_usuario INTEGER NOT NULL,
    id_publicacion INTEGER NOT NULL,
    fecha_prestamo DATE DEFAULT CURRENT_DATE,
    fecha_limite DATE NOT NULL,
    fecha_devolucion DATE,
    estado TEXT DEFAULT 'En curso',
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario) ON DELETE CASCADE,
    FOREIGN KEY (id_publicacion) REFERENCES publicacion(id_publicacion) ON DELETE CASCADE
);

2. Creación de Índices

Índices diseñados para evitar escaneos secuenciales y optimizar tanto las búsquedas de texto como las uniones (JOINs) en llaves foráneas.


-- Índices para búsquedas rápidas de texto
CREATE INDEX idx_libro_titulo ON libro(titulo);
CREATE INDEX idx_lanzamiento_isbn ON lanzamiento_publicacion(isbn);
CREATE INDEX idx_usuario_correo ON usuario(correo);

-- Índices para optimizar los JOINs de las llaves foráneas
CREATE INDEX idx_fk_libro_autor ON libro(id_autor);
CREATE INDEX idx_fk_publicacion_lanzamiento ON publicacion(id_lanzamiento);
CREATE INDEX idx_fk_prestamo_usuario ON prestamo(id_usuario);
CREATE INDEX idx_fk_prestamo_publicacion ON prestamo(id_publicacion);

3. Triggers (Automatización de Inventario)

Disparadores que mantienen la consistencia de los datos entre las tablas de publicaciones físicas y el registro de préstamos.


-- Trigger 1: Evitar que se preste un libro que ya está prestado o en mantenimiento
CREATE TRIGGER trg_validar_disponibilidad
BEFORE INSERT ON prestamo
FOR EACH ROW
BEGIN
    SELECT RAISE(ABORT, 'Error: Esta publicación no está disponible para préstamo.')
    WHERE (SELECT estado FROM publicacion WHERE id_publicacion = NEW.id_publicacion) != 'Disponible';
END;

-- Trigger 2: Cambiar el estado del libro a 'Prestado' al registrar la salida
CREATE TRIGGER trg_marcar_como_prestado
AFTER INSERT ON prestamo
FOR EACH ROW
BEGIN
    UPDATE publicacion 
    SET estado = 'Prestado' 
    WHERE id_publicacion = NEW.id_publicacion;
END;

-- Trigger 3: Regresar el estado a 'Disponible' al devolverlo
CREATE TRIGGER trg_marcar_como_devuelto
AFTER UPDATE OF fecha_devolucion ON prestamo
FOR EACH ROW
WHEN NEW.fecha_devolucion IS NOT NULL AND OLD.fecha_devolucion IS NULL
BEGIN
    UPDATE publicacion 
    SET estado = 'Disponible' 
    WHERE id_publicacion = NEW.id_publicacion;
    
    UPDATE prestamo
    SET estado = 'Devuelto'
    WHERE id_prestamo = NEW.id_prestamo;
END;
